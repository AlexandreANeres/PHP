<?php

echo "Oi, vou ler 3 dados de 5 países diferentes, dados estes que você vai implementar! Boa sorte!";

$paises = array();

for($i = 1; $i < 6; $i++) {
    $nome = readline("\n" . "Nome do País: ")
    $continente = readline("\n" . "Continente em que o país se encontra: ");
    $qtdHabitantes = readline("\n" . "Quantidade de habitantes do país: ");

    $pais["nome"] = $nome;
    $pais["continente"] = $continente;
    $pais["qtdHabitantes"] = $qtdHabitantes;

    array_push($paises, $pais);
}

foreach($paises as $p) {
    echo $p["nome"] . " - " . $p["continente"] . " - " . $p["qtdHabitantes"] . "\n";
}
